import pandas as pd
import ijson
import os

# ---------------------------------------------------------
# CONFIGURATION
# ---------------------------------------------------------
INPUT_FILE = 'sample.json' 

def process_manufacturing_yields(input_path):
    print(f"Processing {input_path}...")
    
    # ---------------------------------------------------------
    # 1. LOAD DATA (Memory Optimized)
    # ---------------------------------------------------------
    data = []
    
    try:
        with open(input_path, 'rb') as f:
            for item in ijson.items(f, 'item'):
                data.append([
                    item.get('partNo'),
                    item.get('uutSn'),
                    item.get('startDateTime'),
                    item.get('productTestPassed'),
                    item.get('ateCpn', 'Unknown'), # Machine Part Number
                    item.get('ateSn', 'Unknown'),  # Machine Serial Number
                    item.get('slotID', -1)         # Slot ID
                ])
                
    except FileNotFoundError:
        print(f"Error: The file '{input_path}' was not found.")
        return

    # Create DataFrame
    cols = ['partNo', 'uutSn', 'startDateTime', 'productTestPassed', 'ateCpn', 'ateSn', 'slotID']
    df = pd.DataFrame(data, columns=cols)
    
    # Memory cleanup
    del data

    # ---------------------------------------------------------
    # 2. PRE-PROCESSING
    # ---------------------------------------------------------
    print("Standardizing dates and sorting...")
    
    # Fix Date Formats (Handle mixed formats)
    df['startDateTime'] = pd.to_datetime(df['startDateTime'], format='mixed', errors='coerce')
    df.dropna(subset=['startDateTime'], inplace=True)

    # SORTING IS CRITICAL
    # We sort by UUT and Time to find the "First Pass" event.
    df.sort_values(by=['uutSn', 'startDateTime'], inplace=True)

    # ---------------------------------------------------------
    # 3. ISOLATE FIRST PASS EVENTS
    # ---------------------------------------------------------
    # We only care about the FIRST time a unit was tested.
    # The machine that tested it first gets the credit (Pass) or the blame (Fail).
    first_attempts = df.drop_duplicates(subset=['partNo', 'uutSn'], keep='first').copy()

    print(f"Total Unique Units Analyzed: {len(first_attempts)}")

    # ---------------------------------------------------------
    # 4. DEFINE CALCULATION FUNCTION
    # ---------------------------------------------------------
    def calculate_yield(df, group_cols, report_name):
        # Group by the specific columns (e.g., Machine SN, or Machine SN + Slot)
        report = df.groupby(group_cols).agg(
            Units_Processed=('uutSn', 'count'),
            Passed_First_Try=('productTestPassed', 'sum')
        ).reset_index()

        # Calculate Yield
        report['Yield_Percentage'] = (report['Passed_First_Try'] / report['Units_Processed']) * 100
        report['Yield_Percentage'] = report['Yield_Percentage'].round(2)
        
        # Sort by Yield (Lowest first - to highlight problems)
        report.sort_values(by='Yield_Percentage', ascending=True, inplace=True)

        print(f"\n--- Report: {report_name} ---")
        print(report.head(3).to_string(index=False))
        
        # Save to CSV
        filename = f"yield_{report_name.lower().replace(' ', '_')}.csv"
        report.to_csv(filename, index=False)
        print(f"Saved to {filename}")

    # ---------------------------------------------------------
    # 5. GENERATE THE 4 REPORTS
    # ---------------------------------------------------------
    
    # Report A: Yield by Machine Part Number (ateCpn)
    # "Is this MODEL of machine performing well?"
    calculate_yield(first_attempts, ['ateCpn'], "By Machine Part")

    # Report B: Yield by Machine Serial Number (ateSn)
    # "Is this SPECIFIC machine broken?"
    calculate_yield(first_attempts, ['ateSn'], "By Machine SN")

    # Report C: Yield by Machine Part + Slot
    # "Is a specific slot on this TYPE of machine problematic?"
    calculate_yield(first_attempts, ['ateCpn', 'slotID'], "By Machine Part and Slot")

    # Report D: Yield by Machine SN + Slot
    # "Is Slot 1 on Machine #006 broken?" (The most granular analysis)
    calculate_yield(first_attempts, ['ateSn', 'slotID'], "By Machine SN and Slot")

if __name__ == "__main__":
    process_manufacturing_yields(INPUT_FILE)
