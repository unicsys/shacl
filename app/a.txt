from pyspark.sql import functions as F, Row


SRC_DB       = "annual_inspection_silver"
SRC_TABLE    = "unique_esn_list"          
SEARCH_DB    = "fst_bronze"                      
RESULT_DB    = "annual_inspection_silver"
RESULT_TABLE = "esn_search_result_fst_bronze"


spark.sql(f"USE {SRC_DB}")
esns = [r.ESN for r in (
    spark.table(f"{SRC_DB}.{SRC_TABLE}")
         .where("ESN IS NOT NULL AND TRIM(ESN) <> ''")
         .select("ESN")
         .distinct()
         .collect()
)]
print(f"Loaded {len(esns)} ESNs")


spark.sql(f"USE {SEARCH_DB}")
tables = [
    r.tableName
    for r in spark.sql(f"SHOW TABLES IN {SEARCH_DB}").collect()
    if (not r.isTemporary) and (not r.tableName.startswith("_"))
]
print(f"Scanning {len(tables)} tables in {SEARCH_DB}")



results = []
for t in tables:
    try:
        df = spark.table(f"{SEARCH_DB}.`{t}`")
        if not df.columns:
            continue
        
        rowjson_df = df.select(
            F.lower(F.to_json(F.struct(*[F.col(c).alias(c) for c in df.columns]))).alias("rowjson")
        )

        for esn in esns:
            cnt = rowjson_df.filter(F.col("rowjson").contains(esn.lower())).count()
            if cnt > 0:
                results.append(Row(table_name=t, esn=esn, matches=int(cnt)))
    except Exception as e:
        print(f"Issue with {t}: {str(e)[:160]}")


if results:
    res_df = spark.createDataFrame(results)

    spark.sql(f"USE {RESULT_DB}")
    res_df.write.mode("overwrite").saveAsTable(f"{RESULT_DB}.{RESULT_TABLE}")

    (
        spark.table(f"{RESULT_DB}.{RESULT_TABLE}")
             .orderBy("table_name", "esn")
             .show(50, truncate=False)
    )

    spark.sql(f"""
      SELECT table_name, SUM(matches) AS total_matches, COUNT(DISTINCT esn) AS esns_hit
      FROM {RESULT_DB}.{RESULT_TABLE}
      GROUP BY table_name
      ORDER BY total_matches DESC
    """).show(100, truncate=False)
else:
    print("No matches")
