db = "engine_demo"

# List all tables in the database
tables = spark.sql(f"SHOW TABLES IN {db}").collect()
table_names = [
    row.tableName
    for row in tables
    if not row.isTemporary and row.tableName.lower() != "_sqldf"
]

# Load DataFrames for each table
dfs = [spark.table(f"{db}.{t}") for t in table_names]

# Union all DataFrames
from functools import reduce
combined_df = reduce(lambda a, b: a.unionByName(b), dfs)

# Save the combined DataFrame as a new Delta table
combined_df.write.format("delta").mode("overwrite").saveAsTable(f"{db}.combined_table")

display(combined_df)
