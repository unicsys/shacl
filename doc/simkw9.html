step_refdes.py import pandas as pd from rdflib import Graph, XSD from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: RefDes] Processing Dynamic Locations ---") g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error: {e}") return # ========================================================= # A. REGENERATE ROW IDs # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= count = 0 for idx, row in df.iterrows(): # Ingredients raw_ref = row.get('RefDesigDescription') # The Location raw_fpn = row.get('FailedComponent') # The Trigger # 1. Child Keys (The Board) raw_pn = row.get('1stLevelMaterial') raw_sn = row.get('1stLevelSerial') # 2. Parent Keys (The System) raw_sys_mat = row.get('Material') raw_sys_ser = row.get('SerialNumber') row_id = row['RowID'] # Normalize key_ref = normalize_key(raw_ref) key_fpn = normalize_key(raw_fpn) key_pn = normalize_key(raw_pn) key_sn = normalize_key(raw_sn) key_sys_mat = normalize_key(raw_sys_mat) key_sys_ser = normalize_key(raw_sys_ser) lit_ref = clean_literal(raw_ref) lit_sn = clean_literal(raw_sn) lit_sys_ser = clean_literal(raw_sys_ser) if not key_ref: continue # ========================================================= # DYNAMIC PARENT LOGIC # ========================================================= parent_iri = None child_iri = None # Keys used for the Location Seed (Must match the Parent) loc_parent_key_1 = "" loc_parent_key_2 = "" if key_fpn: # === SCENARIO A: FAILED COMPONENT EXISTS === # Parent = Board (1st Level) # Child = Electronic Component if not key_pn or not key_sn: continue # 1. Define Parent (Board) parent_iri = generate_iri(CAT_CCA_INST, "GLOBAL", key_pn, key_sn) loc_parent_key_1 = key_pn loc_parent_key_2 = key_sn # 2. Define Child (Component) # Reconstruct Component IRI (Matches Step 5) # Identity: Board + RefDes (or FailedPN) comp_id_key = key_ref CAT_COMPONENT = "electronic_component" child_iri = generate_iri(CAT_COMPONENT, "GLOBAL", key_pn, key_sn, comp_id_key) else: # === SCENARIO B: FAILED COMPONENT EMPTY === # Parent = System (Material/Serial) # Child = Board (1st Level) if not key_sys_mat or not key_sys_ser: continue if not key_pn or not key_sn: continue # Need child board info too # 1. Define Parent (System) parent_iri = generate_iri(CAT_SYS_INST, "GLOBAL", key_sys_mat, key_sys_ser) loc_parent_key_1 = key_sys_mat loc_parent_key_2 = key_sys_ser # 2. Define Child (Board) child_iri = generate_iri(CAT_CCA_INST, "GLOBAL", key_pn, key_sn) # ========================================================= # NODE 1: THE LOCATION (Site) # ========================================================= # Identity: Unique to the *Parent* + RefDes # This ensures R1 on a Board is different from Slot 1 on a System loc_iri = generate_iri(CAT_COMP_LOC, "GLOBAL", loc_parent_key_1, loc_parent_key_2, key_ref) ice_iri = generate_iri(CAT_ID_COMP_LOC, "GLOBAL", key_ref) ibe_iri = generate_iri(CAT_IBE, source_name, row_id, "RefDesigDescription") builder.create_designative_pattern( entity_iri=loc_iri, ice_iri=ice_iri, ibe_iri=ibe_iri, # Label depends on context entity_label=f"Location {lit_ref} on {lit_sn if key_fpn else lit_sys_ser}", ice_label=generate_label("electronic_component_location_identifier", val=lit_ref), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="RefDesigDescription"), literal_value=lit_ref, ice_class=OCKMS.ElectronicComponentLocationIdentifier, data_property=PROP_HAS_TEXT_VALUE, data_type=XSD.string ) builder.add_type(loc_iri, OCKMS.ElectronicComponentLocation) # ========================================================= # CREATE LINKS # ========================================================= # 1. Parent --has_part--> Location builder.add_relationship(parent_iri, REL_HAS_PART, loc_iri) # 2. Child --located_in--> Location builder.add_relationship(child_iri, REL_LOCATED_IN, loc_iri) count += 1 g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} dynamic locations.") print(f"Output saved to: {output_path}")
