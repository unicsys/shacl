run_pipeline.py import os from mappers import ( step_system_identity, step_1stlevelmaterial, step_1revisionlevel, step_1stlevelserial, step_defcode, step_failedcomponent, step_refdes, step_createdon ) # ============================================================================= # CONFIGURATION # ============================================================================= BATCH_ID = "SAP_ZQMDEFREP_NLR_20251007" INPUT_FILE = "input/sample_data.csv" # TOGGLES RUN_STEP_0 = True # System RUN_STEP_1 = True # CCA Designs RUN_STEP_2 = True # CCA Revisions RUN_STEP_3 = True # CCA Instances RUN_STEP_4 = True # Defects RUN_STEP_5 = True # Component Designs RUN_STEP_6 = True # Locations (RefDes) RUN_STEP_7 = True # Processes & Time def main(): if not os.path.exists('output'): os.makedirs('output') print(f"Starting Pipeline: {BATCH_ID}\n") if RUN_STEP_0: step_system_identity.run(INPUT_FILE, "output/00_system_identity.ttl", BATCH_ID) if RUN_STEP_1: step_1stlevelmaterial.run(INPUT_FILE, "output/01_material_designs.ttl", BATCH_ID) if RUN_STEP_2: step_1revisionlevel.run(INPUT_FILE, "output/02_design_revisions.ttl", BATCH_ID) if RUN_STEP_3: step_1stlevelserial.run(INPUT_FILE, "output/03_physical_items.ttl", BATCH_ID) # Note: Steps 4, 5, 6 handle the "Component/Defect Triangle" if RUN_STEP_4: step_defcode.run(INPUT_FILE, "output/04_defects.ttl", BATCH_ID) if RUN_STEP_5: step_failedcomponent.run(INPUT_FILE, "output/05_component_designs.ttl", BATCH_ID) if RUN_STEP_6: step_refdes.run(INPUT_FILE, "output/06_locations.ttl", BATCH_ID) # Step 7: The Event History if RUN_STEP_7: step_createdon.run(INPUT_FILE, "output/07_process_history.ttl", BATCH_ID) print("\nPipeline Complete.") if __name__ == "__main__": main()
