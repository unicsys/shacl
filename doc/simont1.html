step_system_identity.py import pandas as pd from rdflib import Graph, XSD from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: System Identity] Processing System Parent (Material/Serial) ---") # 1. Setup g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) # 2. Load Data try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error reading CSV: {e}") return # ========================================================= # A. GENERATE ROBUST ROW IDs # (Matches exact logic used in CCA steps for consistency) # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = ( df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] ) # Tie-Breaker df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= mapped_systems = set() count = 0 for idx, row in df.iterrows(): # 1. Ingredients raw_mat = row.get('Material') # System Part Number raw_ser = row.get('SerialNumber') # System Serial Number row_id = row['RowID'] key_mat = normalize_key(raw_mat) key_ser = normalize_key(raw_ser) lit_mat = clean_literal(raw_mat) lit_ser = clean_literal(raw_ser) # We need both keys to define a specific physical system instance if not key_mat or not key_ser: continue # --------------------------------------------------------- # NODE 1: SYSTEM DESIGN (The Parent Model) # --------------------------------------------------------- # Identity: Unique per Part Number sys_design_iri = generate_iri(CAT_SYS_DESIGN, "GLOBAL", key_mat) sys_design_ice = generate_iri(CAT_ID_SYS_DES, "GLOBAL", key_mat) sys_design_ibe = generate_iri(CAT_IBE, source_name, row_id, "Material") builder.create_designative_pattern( entity_iri=sys_design_iri, ice_iri=sys_design_ice, ibe_iri=sys_design_ibe, # Label: "System Design {Part}" entity_label=generate_label("system_design", pn=lit_mat), ice_label=generate_label("system_design_identifier", val=lit_mat), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="Material"), literal_value=lit_mat, ice_class=OCKMS.SystemDesignIdentifier, data_property=PROP_HAS_TEXT_VALUE, data_type=XSD.string ) # Deduplicated Type Assertion if key_mat not in mapped_systems: builder.add_type(sys_design_iri, OCKMS.SystemDesign) mapped_systems.add(key_mat) # --------------------------------------------------------- # NODE 2: ENGINEERED SYSTEM (The Physical Instance) # --------------------------------------------------------- # Identity: Unique by Material + Serial sys_inst_iri = generate_iri(CAT_SYS_INST, "GLOBAL", key_mat, key_ser) sys_inst_ice = generate_iri(CAT_ID_SYS_SN, "GLOBAL", key_ser) sys_inst_ibe = generate_iri(CAT_IBE, source_name, row_id, "SerialNumber") builder.create_designative_pattern( entity_iri=sys_inst_iri, ice_iri=sys_inst_ice, ibe_iri=sys_inst_ibe, # Label: "Engineered System {Serial} ({Part})" entity_label=generate_label("engineered_system", pn=lit_mat, sn=lit_ser), ice_label=generate_label("system_serial_number", val=lit_ser), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="SerialNumber"), literal_value=lit_ser, ice_class=OCKMS.SystemSerialNumber, data_property=PROP_HAS_TEXT_VALUE, data_type=XSD.string ) # Type: IOF Engineered System (Using IOF as per your request) builder.add_type(sys_inst_iri, IOF.EngineeredSystem) # --------------------------------------------------------- # LINK: Design --prescribes--> System # --------------------------------------------------------- builder.add_relationship(sys_design_iri, REL_PRESCRIBES, sys_inst_iri) count += 1 # 5. Save g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} system instances.") print(f"Output saved to: {output_path}")
