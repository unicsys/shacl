step_failedcomponent.py import pandas as pd from rdflib import Graph, XSD, Literal, RDF, RDFS, URIRef from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: FailedComponent] Processing Component Designs & Hierarchy ---") g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error: {e}") return # ========================================================= # A. REGENERATE ROW IDs # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= mapped_designs = set() count = 0 for idx, row in df.iterrows(): # Ingredients raw_fpn = row.get('FailedComponent') raw_ref = row.get('RefDesigDescription') # Context raw_pn = row.get('1stLevelMaterial') raw_sn = row.get('1stLevelSerial') row_id = row['RowID'] key_fpn = normalize_key(raw_fpn) key_ref = normalize_key(raw_ref) key_pn = normalize_key(raw_pn) key_sn = normalize_key(raw_sn) lit_fpn = clean_literal(raw_fpn) lit_ref = clean_literal(raw_ref) if not key_fpn: continue # --------------------------------------------------------- # NODE 1: ELECTRONIC COMPONENT DESIGN # --------------------------------------------------------- comp_design_iri = generate_iri(CAT_ELEC_COMP_DESIGN, "GLOBAL", key_fpn) comp_design_ice = generate_iri(CAT_ID_ELEC_COMP_DESIGN, "GLOBAL", key_fpn) comp_design_ibe = generate_iri(CAT_IBE, source_name, row_id, "FailedComponent") builder.create_designative_pattern( entity_iri=comp_design_iri, ice_iri=comp_design_ice, ibe_iri=comp_design_ibe, entity_label=generate_label("electronic_component_design", pn=lit_fpn), ice_label=generate_label("electronic_component_design_identifier", val=lit_fpn), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="FailedComponent"), literal_value=lit_fpn, ice_class=OCKMS.ElectronicComponentDesignIdentifier, data_property=PROP_HAS_TEXT_VALUE, data_type=XSD.string ) if key_fpn not in mapped_designs: builder.add_type(comp_design_iri, OCKMS.ElectronicComponentDesign) mapped_designs.add(key_fpn) # --------------------------------------------------------- # NODE 2: PHYSICAL COMPONENT # --------------------------------------------------------- if key_pn and key_sn: # Identity: Unique to Board + RefDes (or FailedPN if RefDes missing) comp_id_key = key_ref if key_ref else key_fpn physical_comp_iri = generate_iri(CAT_COMPONENT, "GLOBAL", key_pn, key_sn, comp_id_key) # 1. Type it builder.add_type(physical_comp_iri, OCKMS.ElectronicComponent) # 2. Label it (Pass 'pn' so it shows up in the label) # Label = "Electronic Component R143 (705-2437...)" comp_label = generate_label( "electronic_component", ref=lit_ref if lit_ref else lit_fpn, pn=lit_fpn ) builder.g.add((URIRef(physical_comp_iri), RDFS.label, Literal(comp_label))) # 3. Link Design -> Component builder.add_relationship(comp_design_iri, REL_PRESCRIBES, physical_comp_iri) # 4. Link Board -> Component cca_iri = generate_iri(CAT_CCA_INST, "GLOBAL", key_pn, key_sn) builder.add_relationship(cca_iri, REL_HAS_PART, physical_comp_iri) count += 1 g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} component designs (Labels Added).")
