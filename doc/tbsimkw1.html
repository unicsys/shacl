merge.py import os import glob from rdflib import Graph OUTPUT_DIR = "output" MERGED_FILENAME = "FULL_KNOWLEDGE_GRAPH.ttl" def run_merge(): print(f"--- Merging TTL files in {OUTPUT_DIR} ---") # 1. Initialize Master Graph master_g = Graph() # 2. Find all .ttl files # defined order helps debug, but glob is fine too. # We will sort them to ensure deterministic processing order. files = sorted(glob.glob(os.path.join(OUTPUT_DIR, "*.ttl"))) # Avoid re-merging the output file if it already exists files = [f for f in files if MERGED_FILENAME not in f] if not files: print("No TTL files found to merge.") return # 3. Load Loop for file_path in files: try: print(f"Loading: {os.path.basename(file_path)} ...") master_g.parse(file_path, format="turtle") except Exception as e: print(f"ERROR parsing {file_path}: {e}") # 4. Serialize output_path = os.path.join(OUTPUT_DIR, MERGED_FILENAME) print(f"Writing {len(master_g)} triples to {output_path}...") master_g.serialize(destination=output_path, format="turtle") print("Merge Complete.") if __name__ == "__main__": run_merge()
