step_1stlevelserial.py import pandas as pd from rdflib import Graph, XSD from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: 1stLevelSerial] Processing Physical Items & System Links ---") # 1. Setup g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) # 2. Load Data try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error reading CSV: {e}") return # ========================================================= # A. GENERATE ROBUST ROW IDs # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= mapped_items = set() count = 0 for idx, row in df.iterrows(): # 1. Ingredients (CCA - Child) raw_pn = row.get('1stLevelMaterial') raw_sn = row.get('1stLevelSerial') raw_rev = row.get('1RevisionLevel') # 2. Ingredients (System - Parent) raw_sys_mat = row.get('Material') raw_sys_ser = row.get('SerialNumber') row_id = row['RowID'] # Normalize Keys key_pn = normalize_key(raw_pn) key_sn = normalize_key(raw_sn) key_rev = normalize_key(raw_rev) key_sys_mat = normalize_key(raw_sys_mat) key_sys_ser = normalize_key(raw_sys_ser) lit_pn = clean_literal(raw_pn) lit_sn = clean_literal(raw_sn) # We must have CCA Part & Serial to proceed if not key_pn or not key_sn: continue # --------------------------------------------------------- # NODE: PHYSICAL ITEM (The Child) # --------------------------------------------------------- cca_iri = generate_iri(CAT_CCA_INST, "GLOBAL", key_pn, key_sn) ice_iri = generate_iri(CAT_ID_CCA, "GLOBAL", key_sn) ibe_iri = generate_iri(CAT_IBE, source_name, row_id, "1stLevelSerial") builder.create_designative_pattern( entity_iri=cca_iri, ice_iri=ice_iri, ibe_iri=ibe_iri, entity_label=generate_label("circuit_card_assembly", pn=lit_pn, sn=lit_sn), ice_label=generate_label("circuit_card_assembly_identifier", val=lit_sn), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="1stLevelSerial"), literal_value=lit_sn, ice_class=OCKMS.CircuitCardAssemblyIdentifier, data_property=PROP_HAS_TEXT_VALUE, data_type=XSD.string ) # --------------------------------------------------------- # LINKING & LOGIC # --------------------------------------------------------- combo_key = (key_pn, key_sn) if combo_key not in mapped_items: builder.add_type(cca_iri, OCKMS.CircuitCardAssembly) # LINK 1: Revision (or Design) PRESCRIBES CCA if key_rev: rev_iri = generate_iri(CAT_CCA_REV, "GLOBAL", key_pn, key_rev) builder.add_relationship(rev_iri, REL_PRESCRIBES, cca_iri) else: family_iri = generate_iri(CAT_CCA_DESIGN, "GLOBAL", key_pn) builder.add_relationship(family_iri, REL_PRESCRIBES, cca_iri) # LINK 2: System HAS_PART CCA (*** NEW PART ***) if key_sys_mat and key_sys_ser: # We regenerate the System IRI (Matches Step 0) sys_inst_iri = generate_iri(CAT_SYS_INST, "GLOBAL", key_sys_mat, key_sys_ser) # Relation: System --has_continuant_part--> CCA builder.add_relationship(sys_inst_iri, REL_HAS_PART, cca_iri) mapped_items.add(combo_key) count += 1 g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} physical items and linked to Systems.") print(f"Output saved to: {output_path}")
