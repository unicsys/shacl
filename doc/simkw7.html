step_defcode.py import pandas as pd # ENSURE URIRef IS IMPORTED from rdflib import Graph, XSD, Literal, RDF, RDFS, URIRef from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: DefCode] Processing Defects (bearer_of) ---") g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error: {e}") return # ========================================================= # A. REGENERATE ROW IDs # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= count = 0 for idx, row in df.iterrows(): # Ingredients raw_pn = row.get('1stLevelMaterial') raw_sn = row.get('1stLevelSerial') raw_code = row.get('DefCode') raw_desc = row.get('DefCodeDescription') # Target Triggers raw_ref = row.get('RefDesigDescription') raw_fpn = row.get('FailedComponent') row_id = row['RowID'] key_pn = normalize_key(raw_pn) key_sn = normalize_key(raw_sn) key_code = normalize_key(raw_code) key_ref = normalize_key(raw_ref) key_fpn = normalize_key(raw_fpn) lit_code = clean_literal(raw_code) lit_desc = clean_literal(raw_desc) lit_sn = clean_literal(raw_sn) if not key_pn or not key_sn or not key_code: continue # --------------------------------------------------------- # 1. DETERMINE TARGET (Bearer of the Defect) # --------------------------------------------------------- # Default: The Board board_iri = generate_iri(CAT_CCA_INST, "GLOBAL", key_pn, key_sn) target_iri = board_iri # Check if Component Exists (RefDes OR FailedPart) if key_ref or key_fpn: # Reconstruct Component IRI (Matches Step 6 logic) # Identity: Board + RefDes (or FailedPN) comp_id_key = key_ref if key_ref else key_fpn # Use 'electronic_component' category # Ensure CAT_COMPONENT is in config.py, or define it here: # CAT_COMPONENT = "electronic_component" target_iri = generate_iri(CAT_COMPONENT, "GLOBAL", key_pn, key_sn, comp_id_key) # (Note: We assume Step 6 created this node) # --------------------------------------------------------- # 2. CREATE DEFECT (Entity) # --------------------------------------------------------- # Identity: Unique to this Row (Event) defect_iri = generate_iri(CAT_DEFECT, "GLOBAL", row_id, key_code) # ICE: Defect Classification Code ice_iri = generate_iri(CAT_ID_DEFECT, "GLOBAL", key_code) ibe_iri = generate_iri(CAT_IBE, source_name, row_id, "DefCode") builder.create_designative_pattern( entity_iri=defect_iri, ice_iri=ice_iri, ibe_iri=ibe_iri, # Label entity_label=generate_label("non_nominal_quality", code=lit_code, sn=lit_sn), ice_label=generate_label("defect_classification_code", code=lit_code), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="DefCode"), literal_value=lit_code, ice_class=OCKMS.DefectClassificationCode ) builder.add_type(defect_iri, OCKMS.Defect) # --------------------------------------------------------- # 3. LINK: Target --bearer_of--> Defect # --------------------------------------------------------- # Use BFO 'bearer of' (0000196) builder.add_relationship(target_iri, REL_BEARER_OF, defect_iri) # --------------------------------------------------------- # 4. DEFECT DESCRIPTION (Descriptive ICE) # --------------------------------------------------------- if lit_desc: desc_ice_iri = generate_iri(CAT_DESC_DEFECT, "GLOBAL", row_id, key_code) desc_ibe_iri = generate_iri(CAT_IBE, source_name, row_id, "DefCodeDescription") # Manual 'describes' pattern desc_node = builder.g.resource(desc_ice_iri) desc_node.add(RDF.type, OCKMS.DefectClassificationDescription) desc_node.add(RDFS.label, Literal(generate_label("defect_classification_description", code=lit_code))) # Link: Description --describes--> Defect Entity # FIX: URIRef is now imported and will work here desc_node.add(REL_DESCRIBES, URIRef(defect_iri)) # IBE Link ibe_node = builder.g.resource(desc_ibe_iri) ibe_node.add(RDF.type, CCO.InformationBearingEntity) ibe_node.add(RDFS.label, Literal(generate_label("information_bearing_entity", src=source_name, row=row_id, col="DefCodeDescription"))) ibe_node.add(PROP_HAS_TEXT_VALUE, Literal(lit_desc, datatype=XSD.string)) desc_node.add(REL_DEPENDS_ON, ibe_node) count += 1 g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} defects.")
