step_createdon.py import pandas as pd from rdflib import Graph, XSD, Literal, RDF, RDFS, URIRef from framework.config import * from framework.utils import normalize_key, clean_literal, clean_date_time from framework.identity import generate_iri, generate_label from framework.graph_builder import CCOGraphBuilder def run(csv_path, output_path, source_name): print(f"--- [Step: CreatedOn] Processing Processes & Time ---") g = Graph() g.bind("ockms", OCKMS) g.bind("iof", IOF) g.bind("bfo", BFO) g.bind("cco", CCO) builder = CCOGraphBuilder(g) try: df = pd.read_csv(csv_path, low_memory=False) except Exception as e: print(f"Error: {e}") return # ========================================================= # A. REGENERATE ROW IDs # ========================================================= df['clean_date'] = pd.to_datetime(df['CreatedOn'], errors='coerce').dt.strftime('%Y%m%d').fillna('00000000') df['clean_time'] = df['CreatTime'].apply(clean_date_time) df['clean_sn'] = df['1stLevelSerial'].fillna('UNK').astype(str).str.strip().str.upper() df['clean_seq'] = df['SeqNum'].fillna('0').astype(str).str.strip() df['base_id'] = df['clean_sn'] + "_" + df['clean_date'] + "_" + df['clean_time'] + "_" + df['clean_seq'] df['dup_idx'] = df.groupby('base_id').cumcount() df['RowID'] = df['base_id'] + "_" + df['dup_idx'].astype(str) # ========================================================= # B. MAPPING LOOP # ========================================================= count = 0 for idx, row in df.iterrows(): # Ingredients raw_sn = row.get('1stLevelSerial') raw_date = row.get('CreatedOn') raw_code = row.get('DefCode') # Needed to link to Output row_id = row['RowID'] key_sn = normalize_key(raw_sn) key_code = normalize_key(raw_code) # Combine Date+Time timestamp_str = f"{row['clean_date']}-{row['clean_time']}" lit_sn = clean_literal(raw_sn) if not key_sn or not raw_date: continue # --------------------------------------------------------- # NODE 1: INSPECTION PROCESS (Parent) # --------------------------------------------------------- insp_iri = generate_iri(CAT_INSP_PROC, "GLOBAL", row_id) builder.add_type(insp_iri, OCKMS.InspectionProcess) builder.g.add((URIRef(insp_iri), RDFS.label, Literal(generate_label("inspection_process", sn=lit_sn)))) # --------------------------------------------------------- # NODE 2: DEFECT RECORDING PROCESS (Child) # --------------------------------------------------------- rec_iri = generate_iri(CAT_REC_PROC, "GLOBAL", row_id) builder.add_type(rec_iri, OCKMS.DefectRecordingProcess) builder.g.add((URIRef(rec_iri), RDFS.label, Literal(generate_label("defect_recording_process", sn=lit_sn)))) # Link: Inspection --has_occurrent_part--> Recording builder.add_relationship(insp_iri, REL_HAS_OCC_PART, rec_iri) # --------------------------------------------------------- # NODE 3: TEMPORAL INTERVAL (Time) # --------------------------------------------------------- time_iri = generate_iri(CAT_TIME_INT, "GLOBAL", row_id, timestamp_str) time_ice = generate_iri(CAT_ID_TIME, "GLOBAL", timestamp_str) time_ibe = generate_iri(CAT_IBE, source_name, row_id, "CreatedOn_CreatTime") builder.create_designative_pattern( entity_iri=time_iri, ice_iri=time_ice, ibe_iri=time_ibe, entity_label=generate_label("temporal_interval", val=timestamp_str), ice_label=generate_label("temporal_interval_identifier", val=timestamp_str), ibe_label=generate_label("information_bearing_entity", src=source_name, row=row_id, col="CreatedOn"), literal_value=timestamp_str, ice_class=OCKMS.TemporalIntervalIdentifier ) builder.add_type(time_iri, BFO.TemporalInterval) # Link: Recording --occupies--> Time builder.add_relationship(rec_iri, REL_OCCUPIES_TIME, time_iri) # --------------------------------------------------------- # LINK: Inspection --has_output--> Defect Description # --------------------------------------------------------- # (Corrected per Canvas: The Inspection produces the Output) if key_code: # We reconstruct the IRI of the Description created in Step 4 desc_iri = generate_iri(CAT_DESC_DEFECT, "GLOBAL", row_id, key_code) # ** FIX IS HERE **: Use insp_iri instead of rec_iri builder.add_relationship(insp_iri, REL_HAS_OUTPUT, desc_iri) count += 1 g.serialize(destination=output_path, format="turtle") print(f"Success. Mapped {count} process events.")
